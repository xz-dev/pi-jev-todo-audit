const journalReplay = (rows: unknown[]): any[] => rows.map((row) => JSON.parse(JSON.stringify(row)));
const evaluations = (rows: any[]) => rows.filter((row) => row.customType === LEDGER_TYPE && row.data?.kind === "judgment");
const answer = (q: any, choice: string, confidence = 0.9) => {
	if (!Object.hasOwn(q.criteria, choice)) throw new Error(`Fixture chose an undeclared label: ${choice}`);
	return { type: "choice", choice, confidence, probabilities: Object.fromEntries(Object.keys(q.criteria).map((label) => [label, label === choice ? 1 : 0])) };
};

/** Independent decoder of the actual model-facing representation, including current external sources. */
function outboundContext(state: any): any {
	const text = state.fixed.audit as string;
	const start = text.indexOf("\n", text.indexOf("Macro-level evidence")) + 1;
	return JSON.parse(text.slice(start, text.indexOf("\nInterpret evidence", start)));
}
function outboundRecords(state: any): any[] {
	const context = outboundContext(state);
	const external = new Map<string, string>(state.evidence.map((r: any) => [r.id, r.text]));
	return context.records.flatMap((record: any) => {
		if (record.encoding !== "columns/1") return [record];
		expect(new Set(record.columns).size).toBe(record.columns.length);
		return record.rows.map((row: unknown[]) => {
			expect(row.length).toBe(record.columns.length);
			return Object.fromEntries(record.columns.flatMap((column: string, i: number) => row[i] === null ? [] : [[column, row[i]]]));
		});
	}).map((r: any) => ({ ...r, text: external.get(r.id) ?? r.text }));
}

/** Answers depend on actual authoritative goal/permission records, not cached labels or scaffolding. */
function auditResponse(body: any, _raw: string): Response {
	const records = outboundRecords(body.state);
	const user = records.slice().reverse().find((r) => r.kind === "user");
	const goal = records.slice().reverse().find((r) => r.kind === "user" && /GOAL_(XML|CSV)_ONLY/.test(r.text));
	if (!goal) throw new Error("Missing required factual input");
	const reportPresent = records.some((r) => r.kind === "assistant" && r.text.includes("RESULT_XML_ONLY"));
	const choices: Record<string, string> = {
		alignment: "aligned", current_match: "5", drift: "on_track",
		interaction: user?.text.includes("EXECUTION_ALLOWED") ? "working" : "waiting_user", work_evidence: user?.id ?? "insufficient_evidence",
		task_status_5: !reportPresent ? "unclear" : goal.text.includes("GOAL_CSV_ONLY") ? "still_ongoing" : "actually_completed",
		task_evidence_5: reportPresent ? "report" : "insufficient_evidence",
		task_board_5: "accurate", task_granularity_5: "appropriate",
	};
	return Response.json({ answers: Object.fromEntries(Object.entries(body.questions).map(([id, q]) => {
		if (!Object.hasOwn(choices, id)) throw new Error(`Unscripted audit question: ${id}`);
		return [id, answer(q, choices[id])];
	})), usage: { input_tokens: 10 } });
}

async function auditHost(branch: any[], service: Awaited<ReturnType<typeof nativeFixture>>["service"], failPersistence: boolean | ((data: any) => boolean) = false, timeoutMs = 10000) {
	const handlers = new Map<string, any[]>(), commands = new Map<string, any>();
	const sent: any[] = [], notices: string[] = [], reviews: ReviewResult[] = [];
	const original = service.review.bind(service);
	service.review = async (...args) => { const result = await original(...args); reviews.push(structuredClone(result)); return result; };
	globals[key] = service;
	makeExtension({
		on: (name: string, fn: any) => handlers.set(name, [...(handlers.get(name) ?? []), fn]),
		registerCommand: (name: string, command: any) => commands.set(name, command),
		appendEntry: (customType: string, data: unknown) => {
			if (typeof failPersistence === "function" ? failPersistence(data) : failPersistence) throw new Error("consumer disk full");
			branch.push({ id: `audit-${branch.length}`, type: "custom", customType, data });
		},
		sendMessage: (message: any, options: any) => { sent.push({ message, options }); branch.push({ id: `advice-${branch.length}`, type: "custom_message", ...message }); },
		events: { on: () => () => {} },
	} as unknown as ExtensionAPI, { ...DEFAULT_CONFIG, timeoutMs });
	const ctx = { sessionManager: { getSessionId: () => "consumer-cache-acceptance", getBranch: () => branch, buildContextEntries: () => branch }, ui: { notify: (s: string) => notices.push(s) } };
	const emit = async (name: string) => { for (const fn of handlers.get(name) ?? []) await fn({}, ctx); };
	await emit("session_start");
	return { run: (mode = "") => commands.get("jev-audit").handler(mode, ctx), close: () => emit("session_shutdown"), tree: () => emit("session_tree"), reviews, sent, notices };
}

function llmAuditChoice(body: any): string {
	const content = body.messages.slice().reverse().find((m: any) => m.role === "user").content;
	const text = typeof content === "string" ? content : content.filter((b: any) => b.type === "text").map((b: any) => b.text).join("");
	const payload = JSON.parse(text);
	// The same fact-sensitive oracle is used at both real transport boundaries.
	const records = outboundRecords(payload.state);
	const goal = records.slice().reverse().find((r) => r.kind === "user" && /GOAL_(XML|CSV)_ONLY/.test(r.text));
	const user = records.slice().reverse().find((r) => r.kind === "user");
	if (!goal) throw new Error("LLM request omitted the required goal");
	const reported = records.some((r) => r.kind === "assistant" && r.text.includes("RESULT_XML_ONLY"));
	const choices: Record<string, string> = { alignment: "aligned", current_match: "5", drift: "on_track",
		interaction: user?.text.includes("EXECUTION_ALLOWED") ? "working" : "waiting_user", work_evidence: user?.id ?? "insufficient_evidence",
		task_status_5: !reported ? "unclear" : goal.text.includes("GOAL_CSV_ONLY") ? "still_ongoing" : "actually_completed",
		task_evidence_5: reported ? "report" : "insufficient_evidence", task_board_5: "accurate", task_granularity_5: "appropriate" };
	const choice = choices[payload.questionId];
	if (!Object.hasOwn(payload.question.criteria, choice)) throw new Error("Undeclared LLM choice");
	return choice;
}
